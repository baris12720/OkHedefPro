const canvas = document.getElementById("gameCanvas");
const ctx = canvas.getContext("2d");

function resizeCanvas() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
}
window.addEventListener("resize", resizeCanvas);
resizeCanvas();

// Oyun Durum Yönetimi
let score = 0;
let lives = 3;
let wave = 1;
let combo = 0;
let comboTimer = 0;
let gameOver = false;

const targets = [];
const arrows = [];
const particles = [];
const floatingTexts = [];

// Oyuncu Nesnesi (Gelişmiş Görsel Tasarım)
const player = {
    x: 70,
    y: canvas.height / 2,
    radius: 25,
    draw() {
        ctx.save();
        // Yay Gövdesi Efekti
        ctx.beginPath();
        ctx.arc(this.x - 10, this.y, 40, -Math.PI / 2, Math.PI / 2);
        ctx.strokeStyle = "#4fc3f7";
        ctx.lineWidth = 5;
        ctx.stroke();
        
        // Merkez Çekirdek
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fillStyle = "rgba(79, 195, 247, 0.2)";
        ctx.fill();
        ctx.strokeStyle = "#00b0ff";
        ctx.lineWidth = 2;
        ctx.stroke();
        ctx.restore();
    }
};

// Ekran Ortası Koordinat Sabitleme (Mobil için dikey ortalama ayarı)
window.addEventListener("resize", () => {
    player.y = canvas.height / 2;
});

// Partikül (Efekt) Motoru
class Particle {
    constructor(x, y, color) {
        this.x = x;
        this.y = y;
        this.color = color;
        this.radius = Math.random() * 3 + 1;
        this.vx = (Math.random() - 0.5) * 6;
        this.vy = (Math.random() - 0.5) * 6;
        this.alpha = 1;
        this.decay = Math.random() * 0.02 + 0.015;
    }
    update() {
        this.x += this.vx;
        this.y += this.vy;
        this.alpha -= this.decay;
    }
    draw() {
        ctx.save();
        ctx.globalAlpha = this.alpha;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fillStyle = this.color;
        ctx.fill();
        ctx.restore();
    }
}

// Uçan Skor Yazıları Efekti
class FloatingText {
    constructor(x, y, text, color) {
        this.x = x;
        this.y = y;
        this.text = text;
        this.color = color;
        this.vy = -1.5;
        this.alpha = 1;
    }
    update() {
        this.y += this.vy;
        this.alpha -= 0.02;
    }
    draw() {
        ctx.save();
        ctx.globalAlpha = this.alpha;
        ctx.fillStyle = this.color;
        ctx.font = "bold 20px sans-serif";
        ctx.fillText(this.text, this.x, this.y);
        ctx.restore();
    }
}

// Gelişmiş Hedef Sınıfı (Farklı Tipler)
class Target {
    constructor() {
        this.x = canvas.width + 50;
        this.y = Math.random() * (canvas.height - 150) + 75;
        
        const typeRoll = Math.random();
        if (typeRoll < 0.6) {
            // Standart Hedef
            this.type = "normal";
            this.radius = 22;
            this.speed = Math.random() * 1.5 + 2 + (wave * 0.4);
            this.color = "#ff5252"; 
            this.points = 10;
        } else if (typeRoll < 0.85) {
            // Hızlı Hedef
            this.type = "fast";
            this.radius = 14;
            this.speed = Math.random() * 2 + 4 + (wave * 0.5);
            this.color = "#ffd740";
            this.points = 25;
        } else {
            // Zırhlı Büyük Boss Hedef
            this.type = "boss";
            this.radius = 35;
            this.speed = 1.2 + (wave * 0.2);
            this.color = "#e040fb";
            this.points = 50;
        }
    }
    update() {
        this.x -= this.speed;
    }
    draw() {
        ctx.save();
        ctx.shadowBlur = 10;
        ctx.shadowColor = this.color;
        
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fillStyle = this.color;
        ctx.fill();
        
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius * 0.5, 0, Math.PI * 2);
        ctx.fillStyle = "#ffffff";
        ctx.fill();
        
        ctx.restore();
    }
}

// Gelişmiş Ok Sınıfı
class Arrow {
    constructor(x, y) {
        this.x = x;
        this.y = y;
        this.speed = 14;
        this.width = 35;
    }
    update() {
        this.x += this.speed;
    }
    draw() {
        ctx.save();
        let gradient = ctx.createLinearGradient(this.x, this.y, this.x + this.width, this.y);
        gradient.addColorStop(0, "rgba(0, 229, 255, 0)");
        gradient.addColorStop(1, "rgba(0, 229, 255, 1)");
        
        ctx.beginPath();
        ctx.moveTo(this.x, this.y);
        ctx.lineTo(this.x + this.width, this.y);
        ctx.strokeStyle = gradient;
        ctx.lineWidth = 4;
        ctx.stroke();
        ctx.restore();
    }
}

function fire() {
    if (gameOver) {
        resetGame();
        return;
    }
    arrows.push(new Arrow(player.x + 10, player.y));
}

window.addEventListener("click", fire);
window.addEventListener("touchstart", (e) => {
    e.preventDefault();
    fire();
});

function createExplosion(x, y, color) {
    for (let i = 0; i < 15; i++) {
        particles.push(new Particle(x, y, color));
    }
}

function resetGame() {
    score = 0;
    lives = 3;
    wave = 1;
    combo = 0;
    gameOver = false;
    targets.length = 0;
    arrows.length = 0;
    particles.length = 0;
    floatingTexts.length = 0;
    loop();
}

function loop() {
    if (gameOver) {
        ctx.fillStyle = "rgba(11, 14, 20, 0.85)";
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        
        ctx.fillStyle = "#ff5252";
        ctx.font = "bold 36px sans-serif";
        ctx.textAlign = "center";
        ctx.fillText("OYUN BİTTİ", canvas.width / 2, canvas.height / 2 - 40);
        
        ctx.fillStyle = "#ffffff";
        ctx.font = "22px sans-serif";
        ctx.fillText(`Toplam Skor: ${score}  |  Wave: ${wave}`, canvas.width / 2, canvas.height / 2 + 10);
        
        ctx.fillStyle = "#b0bec5";
        ctx.font = "16px sans-serif";
        ctx.fillText("Devam etmek için ekrana dokunun", canvas.width / 2, canvas.height / 2 + 60);
        return;
    }

    ctx.fillStyle = "#0b0e14";
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    ctx.strokeStyle = "rgba(255, 255, 255, 0.02)";
    ctx.lineWidth = 1;
    for (let x = 0; x < canvas.width; x += 40) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, canvas.height); ctx.stroke();
    }

    if (comboTimer > 0) {
        comboTimer--;
        if (comboTimer === 0) combo = 0;
    }

    wave = Math.floor(score / 150) + 1;

    player.draw();

    let spawnChance = 0.015 + (wave * 0.003);
    if (Math.random() < Math.min(spawnChance, 0.05) && targets.length < 6) {
        targets.push(new Target());
    }

    for (let i = arrows.length - 1; i >= 0; i--) {
        arrows[i].update();
        arrows[i].draw();
        if (arrows[i].x > canvas.width) arrows.splice(i, 1);
    }

    for (let i = targets.length - 1; i >= 0; i--) {
        targets[i].update();
        targets[i].draw();

        if (targets[i].x < 0) {
            targets.splice(i, 1);
            lives--;
            combo = 0;
            if (lives <= 0) gameOver = true;
            continue;
        }

        for (let j = arrows.length - 1; j >= 0; j--) {
            if (targets[i] && arrows[j]) {
                let dist = Math.hypot(arrows[j].x + arrows[j].width - targets[i].x, arrows[j].y - targets[i].y);
                if (dist < targets[i].radius) {
                    combo++;
                    comboTimer = 120;
                    let currentGain = targets[i].points * (combo > 1 ? combo : 1);
                    score += currentGain;

                    createExplosion(targets[i].x, targets[i].y, targets[i].color);
                    floatingTexts.push(new FloatingText(targets[i].x, targets[i].y, `+${currentGain}`, targets[i].color));
                    if (combo > 1) {
                        floatingTexts.push(new FloatingText(targets[i].x, targets[i].y - 25, `${combo}x Combo!`, "#ffd740"));
                    }

                    targets.splice(i, 1);
                    arrows.splice(j, 1);
                    break;
                }
            }
        }
    }

    for (let i = particles.length - 1; i >= 0; i--) {
        particles[i].update();
        particles[i].draw();
        if (particles[i].alpha <= 0) particles.splice(i, 1);
    }
    for (let i = floatingTexts.length - 1; i >= 0; i--) {
        floatingTexts[i].update();
        floatingTexts[i].draw();
        if (floatingTexts[i].alpha <= 0) floatingTexts.splice(i, 1);
    }

    ctx.fillStyle = "#ffffff";
    ctx.font = "bold 20px sans-serif";
    ctx.textAlign = "left";
    ctx.fillText(`SKOR: ${score}`, 20, 40);
    ctx.fillText(`WAVE: ${wave}`, 180, 40);

    ctx.textAlign = "right";
    ctx.fillStyle = "#b0bec5";
    ctx.font = "16px sans-serif";
    ctx.fillText("CAN: ", canvas.width - 110, 38);
    for (let i = 0; i < 3; i++) {
        ctx.fillStyle = i < lives ? "#ff5252" : "rgba(255,82,82,0.2)";
        ctx.fillRect(canvas.width - 100 + (i * 25), 24, 18, 18);
    }

    requestAnimationFrame(loop);
}

loop();

name: Build Android APK
on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Source Code
        uses: actions/checkout@v3

      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-size: 16

      - name: Install Cordova
        run: npm install -g cordova

      - name: Create Cordova Project
        run: |
          cordova create build_folder com.okhedef.pro OkHedefPro
          cd build_folder
          cordova platform add android
          rm -rf www/*
          
      - name: Copy Game Files
        run: |
          cp index.html build_folder/www/
          cp game.js build_folder/www/

      - name: Setup Java JDK
        uses: actions/setup-java@v3
        with:
          distribution: 'zulu'
          java-version: '11'

      - name: Build Android APK
        run: |
          cd build_folder
          cordova build android --no-telemetry

      - name: Upload APK Application
        uses: actions/upload-artifact@v3
        with:
          name: OkHedefPro-Game-APK
          path: build_folder/platforms/android/app/build/outputs/apk/debug/app-debug.apk
